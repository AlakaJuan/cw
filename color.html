Google Apps Script (GAS) 部署為網頁應用程式 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    
    // Append to sheet: [Timestamp, HEX, RGB, Source]
    sheet.appendRow([timestamp, data.hex, data.rgb, data.source]);
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

Python Flask app.py 

from flask import Flask, request, jsonify
import json
from datetime import datetime

app = Flask(__name__)

@app.route('/api/save-color', methods=['POST'])
def save_color():
    data = request.json
    record = {
        "timestamp": datetime.now().isoformat(),
        "hex": data.get("hex"),
        "rgb": data.get("rgb"),
        "source": data.get("source", "web")
    }
    
    # Save to JSON file
    with open('colors.json', 'a') as f:
        f.write(json.dumps(record) + '\n')
        
    return jsonify({"status": "success", "message": "Color saved."})

if __name__ == '__main__':
    app.run(debug=True, port=5000)
